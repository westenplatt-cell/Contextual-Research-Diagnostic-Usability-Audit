## 2. Add the JavaScript

Put this in a file like `scripts/grade-sort.js`, or inside a `<script>` tag at the bottom of your grade page.

This script will:

- Sort by due date and add a divider for each unique date.
- Show only assignments for Assignment Group.
- Show only quizzes for Quiz Group.
- Group items by module.
- Sort alphabetically by name.
- Re-render the table whenever the dropdown changes.

```

const gradeSort = document.querySelector("#grade-sort");
const gradeTableBody = document.querySelector("#grade-table-body");

// Keep the original assignment rows so we can re-sort them.
const gradeRows = Array.from(
    gradeTableBody.querySelectorAll("tr[data-type]")
);

// Create a divider row for the table.
function createDivider(label) {
    const divider = document.createElement("tr");
    divider.className = "sort-divider";

    const cell = document.createElement("td");
    cell.colSpan = 8;
    cell.textContent = label;

    divider.appendChild(cell);

    return divider;
}

// Group rows by a particular property.
function groupRows(rows, property) {
    return rows.reduce((groups, row) => {
        const key = row.dataset[property] || "Uncategorized";

        if (!groups[key]) {
            groups[key] = [];
        }

        groups[key].push(row);

        return groups;
    }, {});
}

// Display rows and optional divider rows.
function renderGroups(groups, labelPrefix = "") {
    gradeTableBody.replaceChildren();

    Object.entries(groups).forEach(([groupName, rows]) => {
        gradeTableBody.appendChild(
            createDivider(`${labelPrefix}${groupName}`)
        );

        rows.forEach((row) => {
            gradeTableBody.appendChild(row);
        });
    });
}

// Handle each dropdown option.
function sortGrades() {
    const selectedSort = gradeSort.value;

    // Start from the original rows each time.
    let rows = [...gradeRows];

    switch (selectedSort) {

        case "due-date": {
            rows.sort((a, b) =>
                a.dataset.due.localeCompare(b.dataset.due)
            );

            const groups = groupRows(rows, "due");

            renderGroups(groups, "Due: ");
            break;
        }

        case "assignment": {
            rows = rows.filter(
                (row) => row.dataset.type === "assignment"
            );

            rows.sort((a, b) =>
                a.dataset.name.localeCompare(b.dataset.name)
            );

            gradeTableBody.replaceChildren();

            rows.forEach((row) => {
                gradeTableBody.appendChild(row);
            });

            break;
        }

        case "quiz": {
            rows = rows.filter(
                (row) => row.dataset.type === "quiz"
            );

            rows.sort((a, b) =>
                a.dataset.name.localeCompare(b.dataset.name)
            );

            gradeTableBody.replaceChildren();

            rows.forEach((row) => {
                gradeTableBody.appendChild(row);
            });

            break;
        }

        case "module": {
            const groups = groupRows(rows, "module");

            // Sort the module names alphabetically.
            const sortedGroups = Object.fromEntries(
                Object.entries(groups).sort(([a], [b]) =>
                    a.localeCompare(b)
                )
            );

            renderGroups(sortedGroups, "");
            break;
        }

        case "name": {
            rows.sort((a, b) =>
                a.dataset.name.localeCompare(b.dataset.name)
            );

            gradeTableBody.replaceChildren();

            rows.forEach((row) => {
                gradeTableBody.appendChild(row);
            });

            break;
        }
    }
}
```
